toInteger(floor(80*rand())) AS color_position,
toInteger(floor(25*rand())) AS nation_position
WITH nations[nation_position] AS nation,
colors[color_position] AS color,
date('1993-01-01') + Duration({years: toInteger(floor(4*rand()))}) AS start_date
MATCH (s:SUPPLIER), (n:NATION), (ps:PARTSUPP), (p:PART), (l:LINEITEM)
WHERE n.n_name = nation
  AND s.s_nationkey = n.n_nationkey 
  AND ps.ps_suppkey = s.s_suppkey  
  AND ps.ps_partkey = p.p_partkey  
  AND l.l_suppkey = s.s_suppkey  
  AND l.l_partkey = p.p_partkey  
  AND p.p_name CONTAINS color  
  AND l.l_shipdate >= start_date
  AND l.l_shipdate < start_date + Duration({years: 1})
WITH s, p, SUM(l.l_quantity) AS total_qty
WHERE total_qty <= 300
RETURN s.s_name AS supplier_name, s.s_address AS supplier_address
ORDER BY s.s_name
