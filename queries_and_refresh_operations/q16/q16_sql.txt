# parameters

p_brand, p_type, p_size


# query with flexible parameters

SET @types1 = JSON_ARRAY('STANDARD', 'SMALL', 'MEDIUM', 'LARGE', 'ECONOMY', 'PROMO');
SET @type1 = JSON_UNQUOTE(JSON_EXTRACT(@types1, CONCAT('$[',FLOOR(5*RAND()),']')));
SET @types2 = JSON_ARRAY('ANODIZED', 'BURNISHED', 'PLATED', 'POLISHED', 'BRUSHED');
SET @type2 = JSON_UNQUOTE(JSON_EXTRACT(@types2, CONCAT('$[',FLOOR(4*RAND()),']')));
SET @type = CONCAT(@type1, ' ', @type2);
SET @digit1 = FLOOR(5*RAND());
SET @digit2 = FLOOR(5*RAND());
select
  p_brand,
  p_type,
  p_size,
  count(distinct ps_suppkey) as supplier_cnt
from
  partsupp,
  part
where
  p_partkey = ps_partkey
  and p_brand <> CONCAT('Brand#', @digit1, @digit2)
  and p_type not like CONCAT(@type, '%')
  and p_size in (
  select random from (select distinct floor(1+ 49*rand()) as random from customer LIMIT 8) AS rands
  )
  and ps_suppkey not in (
    select
      s_suppkey
    from
      supplier
    where
      s_comment like '%Customer%Complaints%'
  )
group by
  p_brand,
  p_type,
  p_size
order by
  supplier_cnt desc,
  p_brand,
  p_type,
  p_size;





# validation query


select
  p_brand,
  p_type,
  p_size,
  count(distinct ps_suppkey) as supplier_cnt
from
  partsupp,
  part
where
  p_partkey = ps_partkey
  and p_brand <> 'Brand#45'
  and p_type not like 'MEDIUM POLISHED%'
  and p_size in (49, 14, 23, 45, 19, 3, 36, 9)
  and ps_suppkey not in (
    select
      s_suppkey
    from
      supplier
    where
      s_comment like '%Customer%Complaints%'
  )
group by
  p_brand,
  p_type,
  p_size
order by
  supplier_cnt desc,
  p_brand,
  p_type,
  p_size;
