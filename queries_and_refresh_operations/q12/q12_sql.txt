# parameters

l_shipmode, l_receiptdate



# query with flexible parameters

SET @startDate = date('1993-01-01') + interval floor(4*rand()) year;
SET @modes = JSON_ARRAY('REG AIR', 'AIR', 'RAIL', 'SHIP', 'TRUCK', 'MAIL', 'FOB');
SET @modePosition1 = FLOOR(7*RAND());
SET @modePosition1 = MOD(@modePosition1 + FLOOR(5*RAND()), 7);
SET @mode1 = JSON_UNQUOTE(JSON_EXTRACT(@modes, CONCAT('$[', @modePosition1,']')));
SET @mode2 = JSON_UNQUOTE(JSON_EXTRACT(@modes, CONCAT('$[', @modePosition2,']')));
select
  l_shipmode,
  sum(case
    when o_orderpriority = '1-URGENT'
      or o_orderpriority = '2-HIGH'
      then 1
    else 0
  end) as high_line_count,
  sum(case
    when o_orderpriority <> '1-URGENT'
      and o_orderpriority <> '2-HIGH'
      then 1
    else 0
  end) as low_line_count
from
  orders,
  lineitem
where
  o_orderkey = l_orderkey
  and l_shipmode in (@mode1, @mode2)
  and l_commitdate < l_receiptdate
  and l_shipdate < l_commitdate
  and l_receiptdate >= @startDate
  and l_receiptdate < @startDate + interval '1' year
group by
  l_shipmode
order by
  l_shipmode;




# validation query

select
  l_shipmode,
  sum(case
    when o_orderpriority = '1-URGENT'
      or o_orderpriority = '2-HIGH'
      then 1
    else 0
  end) as high_line_count,
  sum(case
    when o_orderpriority <> '1-URGENT'
      and o_orderpriority <> '2-HIGH'
      then 1
    else 0
  end) as low_line_count
from
  orders,
  lineitem
where
  o_orderkey = l_orderkey
  and l_shipmode in ('MAIL', 'SHIP')
  and l_commitdate < l_receiptdate
  and l_shipdate < l_commitdate
  and l_receiptdate >= date '1994-01-01'
  and l_receiptdate < date '1994-01-01' + interval '1' year
group by
  l_shipmode
order by
  l_shipmode;
